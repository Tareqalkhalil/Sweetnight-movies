/**
 * بناء رابط مع فلاتر
 */
function buildUrl($base, $filters = []) {
    $params = [];
    
    if (!empty($filters['genre'])) {
        $params[] = 'genre=' . urlencode($filters['genre']);
    }
    if (!empty($filters['quality'])) {
        $params[] = 'quality=' . urlencode($filters['quality']);
    }
    if (!empty($filters['year'])) {
        $params[] = 'year=' . intval($filters['year']);
    }
    if (!empty($filters['country'])) {
        $params[] = 'country=' . urlencode($filters['country']);
    }
    if (!empty($filters['language'])) {
        $params[] = 'language=' . urlencode($filters['language']);
    }
    
    if (!empty($params)) {
        return $base . '?' . implode('&', $params);
    }
    
    return $base;
}

/**
 * جلب محتوى الصفحة
 */
function fetchPage($url) {
    $ch = curl_init();
    curl_setopt($ch, CURLOPT_URL, $url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
    curl_setopt($ch, CURLOPT_TIMEOUT, 30);
    curl_setopt($ch, CURLOPT_USERAGENT, 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36');
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    
    $html = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    
    if ($httpCode !== 200 || empty($html)) {
        return false;
    }
    
    return $html;
}

/**
 * استخراج الأفلام/المسلسلات من HTML
 */
function parseContent($html) {
    $items = [];
    
    // البحث عن العناصر باستخدام regex
    // نمط العنوان والرابط
    preg_match_all('/<a[^>]*href=["\']([^"\']*\/(?:movie|series|anime)\/[^"\']*)["\'][^>]*>.*?<h3[^>]*>(.*?)<\/h3>/si', $html, $matches, PREG_SET_ORDER);
    
    foreach ($matches as $match) {
        $items[] = [
            'url' => $match[1],
            'title' => strip_tags($match[2]),
        ];
    }
    
    // استخراج الصورة
    preg_match_all('/<img[^>]*src=["\']([^"\']*)["\'][^>]*alt=["\']([^"\']*)["\'][^>]*>/si', $html, $imgMatches, PREG_SET_ORDER);
    
    // استخراج التقييم
    preg_match_all('/(\d+\.\d+)<\/span>/', $html, $ratingMatches);
    
    // استخراج الجودة
    preg_match_all('/(1080p|720p|480p)[\s\-]*(BluRay|WEB-DL|WEBRip|HDCAM|HDTS|HDTV|WEB|HDRip|HDTC|WEBSCR)/i', $html, $qualityMatches);
    
    // استخراج النوع
    preg_match_all('/>(اكشن|دراما|رعب|رومانسي|كوميدي|اثارة|جريمة|خيال علمي|فانتازيا|مغامرة|حربي|تاريخي|عائلي|غموض|كرتون|موسيقي|وثائقي|ويسترن|رياضي|سيرة ذاتية|قصير|game-show|reality-tv|talk-show|music)</i', $html, $genreMatches);
    
    return [
        'items' => $items,
        'images' => $imgMatches,
        'ratings' => $ratingMatches[1] ?? [],
        'qualities' => $qualityMatches[0] ?? [],
        'genres' => $genreMatches[1] ?? [],
    ];
}

/**
 * الحصول على رابط الصفحة التالية
 */
function getNextPageUrl($html, $currentUrl) {
    // البحث عن رابط "التالي" أو "Next"
    if (preg_match('/<a[^>]*href=["\']([^"\']*(?:page\/\d+|page=\d+|paged=\d+)[^"\']*)["\'][^>]*>(?:التالي|Next|›|»)/i', $html, $match)) {
        return $match[1];
    }
    
    // إذا كان الرابط يحتوي على ?page=
    if (preg_match('/page=(\d+)/', $currentUrl, $match)) {
        $nextPage = intval($match[1]) + 1;
        return preg_replace('/page=\d+/', 'page=' . $nextPage, $currentUrl);
    }
    
    return false;
}

/**
 * استيراد المحتوى
 */
function importContent($urls, $maxPages = 1) {
    $allItems = [];
    
    foreach ($urls as $name => $url) {
        echo "جاري استيراد: $name\n";
        
        $currentUrl = $url;
        $pageCount = 0;
        
        while ($currentUrl && $pageCount < $maxPages) {
            $html = fetchPage($currentUrl);
            
            if (!$html) {
                echo "فشل في جلب: $currentUrl\n";
                break;
            }
            
            $data = parseContent($html);
            $allItems[$name] = array_merge($allItems[$name] ?? [], $data['items']);
            
            echo "تم استخراج " . count($data['items']) . " عنصر من الصفحة\n";
            
            $currentUrl = getNextPageUrl($html, $currentUrl);
            $pageCount++;
            
            // تأخير لتجنب الحظر
            sleep(2);
        }
    }
    
    return $allItems;
}

/**
 * حفظ في قاعدة البيانات (WordPress مثال)
 */
function saveToDatabase($items, $postType = 'post') {
    global $wpdb;
    
    foreach ($items as $category => $categoryItems) {
        foreach ($categoryItems as $item) {
            // التحقق من عدم التكرار
            $existing = $wpdb->get_var($wpdb->prepare(
                "SELECT ID FROM {$wpdb->posts} WHERE post_title = %s AND post_type = %s",
                $item['title'],
                $postType
            ));
            
            if ($existing) {
                echo "تخطي (موجود مسبقاً): {$item['title']}\n";
                continue;
            }
            
            // إنشاء المنشور
            $postData = [
                'post_title'   => $item['title'],
                'post_content' => '',
                'post_status'  => 'publish',
                'post_type'    => $postType,
                'post_author'  => 1,
            ];
            
            $postId = wp_insert_post($postData);
            
            if ($postId) {
                // إضافة حقول مخصصة
                update_post_meta($postId, 'source_url', $item['url']);
                update_post_meta($postId, 'import_category', $category);
                
                echo "تم استيراد: {$item['title']}\n";
            }
        }
    }
}

// معالجة النموذج
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    header('Content-Type: text/plain; charset=utf-8');
    
    $selectedSections = $_POST['sections'] ?? [];
    $selectedGenres = $_POST['genres'] ?? [];
    $selectedQualities = $_POST['qualities'] ?? [];
    $selectedYear = $_POST['year'] ?? '';
    $selectedCountry = $_POST['country'] ?? '';
    $selectedLanguage = $_POST['language'] ?? '';
    $maxPages = intval($_POST['max_pages'] ?? 5);
    
    if (empty($selectedSections)) {
        die("❌ الرجاء اختيار قسم واحد على الأقل!");
    }
    
    // بناء قائمة الروابط
    $urlsToImport = [];
    
    foreach ($selectedSections as $section) {
        $baseUrl = $categories[$section] ?? '';
        if (empty($baseUrl)) continue;
        
        // إذا لم يتم اختيار فلاتر إضافية
        if (empty($selectedGenres) && empty($selectedQualities) && empty($selectedYear) && empty($selectedCountry) && empty($selectedLanguage)) {
            $urlsToImport[$section] = $baseUrl;
        } else {
            // توليد كل التركيبات الممكنة
            $combinations = [[]];
            
            if (!empty($selectedGenres)) {
                $newCombinations = [];
                foreach ($combinations as $comb) {
                    foreach ($selectedGenres as $genre) {
                        $newCombinations[] = array_merge($comb, ['genre' => $genre]);
                    }
                }
                $combinations = $newCombinations;
            }
            
            if (!empty($selectedQualities)) {
                $newCombinations = [];
                foreach ($combinations as $comb) {
                    foreach ($selectedQualities as $quality) {
                        $newCombinations[] = array_merge($comb, ['quality' => $quality]);
                    }
                }
                $combinations = $newCombinations;
            }
            
            if (!empty($selectedYear)) {
                foreach ($combinations as &$comb) {
                    $comb['year'] = $selectedYear;
                }
            }
            
            if (!empty($selectedCountry)) {
                foreach ($combinations as &$comb) {
                    $comb['country'] = $selectedCountry;
                }
            }
            
            if (!empty($selectedLanguage)) {
                foreach ($combinations as &$comb) {
                    $comb['language'] = $selectedLanguage;
                }
            }
            
            foreach ($combinations as $comb) {
                $urlKey = $section . '_' . md5(serialize($comb));
                $urlsToImport[$urlKey] = buildUrl($baseUrl, $comb);
            }
        }
    }
    
    echo "📋 الروابط التي سيتم استيرادها:\n";
    echo str_repeat("=", 80) . "\n";
    foreach ($urlsToImport as $key => $url) {
        echo "[$key] => $url\n";
    }
    echo str_repeat("=", 80) . "\n\n";
    
    // بدء الاستيراد
    echo "⏳ جاري الاستيراد...\n\n";
    
    $results = importContent($urlsToImport, $maxPages);
    
    echo "\n✅ تم الانتهاء!\n";
    echo "📊 الإحصائيات:\n";
    foreach ($results as $key => $items) {
        echo "  - $key: " . count($items) . " عنصر\n";
    }
    
    // حفظ النتائج في ملف JSON للمراجعة
    $jsonFile = 'import_results_' . date('Y-m-d_H-i-s') . '.json';
    file_put_contents($jsonFile, json_encode($results, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
    echo "\n💾 تم حفظ النتائج في: $jsonFile\n";
}
